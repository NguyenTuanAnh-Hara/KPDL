% 5. THUẬT TOÁN
% =====================================================================
\begin{frame}{Thuật toán KNN: các bước}
\begin{cbox}[navy]{Bốn bước dự đoán cho một điểm mới $x$}
\centering
\begin{tikzpicture}[
  st/.style={draw=navy, fill=navy!6, rounded corners=2pt, align=center, font=\scriptsize,
             minimum height=1.2cm, text width=2.2cm},
  ar/.style={-{Stealth[length=2mm]}, thick, navy}]
  \node[st, fill=black!5, draw=black!50] (d) {Tập huấn luyện\\$D=\{(x_i,y_i)\}$\\+ điểm mới $x$};
  \node[st, right=0.3cm of d] (s1) {\textbf{B1.} Chọn K và\\hàm khoảng cách;\\chuẩn hóa dữ liệu};
  \node[st, right=0.3cm of s1] (s2) {\textbf{B2.} Tính\\$d(x,x_i)$\\với mọi $i=1..n$};
  \node[st, right=0.3cm of s2] (s3) {\textbf{B3.} Sắp xếp,\\lấy K điểm gần nhất\\$N_K(x)$};
  \node[st, right=0.3cm of s3] (s4) {\textbf{B4.} Bỏ phiếu đa số\\(hồi quy: lấy\\trung bình)};
  \foreach \a/\b in {d/s1,s1/s2,s2/s3,s3/s4} \draw[ar] (\a) -- (\b);
\end{tikzpicture}
\end{cbox}
\vspace{0.2em}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.48\textwidth}
\begin{cbox}[teal]{Công thức dự đoán}
  Phân lớp: \quad $\displaystyle \hat y=\arg\max_{c}\sum_{x_i\in N_K(x)} \mathbb{I}(y_i=c)$\\[0.3em]
  Hồi quy: \quad $\displaystyle \hat y=\frac1K\sum_{x_i\in N_K(x)} y_i$\\[0.3em]
  Có trọng số: \ $w_i = 1/d(x,x_i)$ \ (Dudani, 1976)
\end{cbox}
\end{column}
\begin{column}{0.48\textwidth}
\begin{cbox}[orange]{Độ đo khoảng cách thường dùng}
\footnotesize
\renewcommand{\arraystretch}{1.9}
\begin{tabular}{@{}ll@{}}
Euclid & $\sqrt{\textstyle\sum_j (x_j-z_j)^2}$\\
Manhattan & $\textstyle\sum_j |x_j-z_j|$\\
Chebyshev & $\max_j |x_j-z_j|$\\
Cosine & $1-(x\cdot z)\,/\,(\|x\|\,\|z\|)$
\end{tabular}
\end{cbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Cài đặt KNN từ đầu (TN1)}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.58\textwidth}
\begin{cbox}[navy]{\texttt{KNNScratch} -- đúng 4 bước của thuật toán}
\footnotesize
\begin{verbatim}
class KNNScratch:
    def __init__(self, k=5):
        self.k = k
    def fit(self, X, y):        # chỉ lưu dữ liệu
        self.X, self.y = np.asarray(X), np.asarray(y)
        return self
    def _predict_one(self, x):
        d = np.sqrt(((self.X - x) ** 2).sum(axis=1))
        idx = np.argsort(d)[: self.k]
        return Counter(self.y[idx]).most_common(1)[0][0]
\end{verbatim}
\end{cbox}
\end{column}
\begin{column}{0.39\textwidth}
\begin{cbox}[green]{Kết quả kiểm chứng (K = 5)}
  Trùng khớp \hl{114/114} dự đoán với \texttt{KNeighborsClassifier} của scikit-learn\\[0.2em]
  Cùng accuracy \hl{95,61\%} trên tập test
\end{cbox}
\begin{cbox}[orange]{Pipeline chuẩn}
  \texttt{StandardScaler} $\rightarrow$ \texttt{KNN}\\
  đặt trong \texttt{cross\_val\_score} để chuẩn hóa được fit lại trên từng fold, \textbf{tránh rò rỉ dữ liệu}
\end{cbox}
